cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(KeyboardMouseRemapper VERSION 1.0.3 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
 message(FATAL_ERROR "Use the Visual Studio 2022 x64 C++ toolchain on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX GR_STATIC)
add_compile_options(/W4 /WX /utf-8 /permissive- /guard:cf /EHsc)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT "/PDBALTPATH:%_PDB%")
option(GPR_ASAN "Instrument application and tests with AddressSanitizer" OFF)
if(GPR_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
 set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
 set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(RemapperCore STATIC App/Core/ControlFrame.cpp App/Core/ScriptCompiler.cpp App/Core/ScriptInstance.cpp App/Core/VisualDocument.cpp)
target_link_libraries(RemapperCore PUBLIC bcrypt)
add_library(RemapperEngine STATIC App/Capture/InputEngine.cpp)
target_link_libraries(RemapperEngine PUBLIC RemapperCore)
target_include_directories(RemapperEngine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
add_library(RemapperCapture STATIC App/Capture/Service.cpp App/Capture/DriverTransport.cpp App/Capture/LiveTests.cpp)
target_link_libraries(RemapperCapture PUBLIC RemapperEngine InterceptorSDK)
file(GLOB ui_sources CONFIGURE_DEPENDS App/UI/*.cpp)
add_library(RemapperUI STATIC ${ui_sources})
target_link_libraries(RemapperUI PUBLIC RemapperCapture comctl32 comdlg32 gdi32 shell32 ole32 oleaut32 oleacc uuid dwmapi uxtheme windowscodecs version)
add_executable(KeyboardMouseRemapper WIN32 App/Main.cpp App/App.rc)
target_link_libraries(KeyboardMouseRemapper PRIVATE RemapperUI)
target_link_options(KeyboardMouseRemapper PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/App/App.manifest")
enable_testing()
foreach(suite Core Engine Worker)
 add_executable(Remapper${suite}Tests App/Tests/${suite}Tests.cpp)
 target_link_libraries(Remapper${suite}Tests PRIVATE RemapperCapture)
 add_test(NAME ${suite} COMMAND Remapper${suite}Tests "${CMAKE_SOURCE_DIR}/Scripts")
endforeach()
foreach(suite Widget ScriptPanel ScriptLibraryPopup)
 add_executable(Remapper${suite}Tests App/Tests/${suite}Tests.cpp)
 target_link_libraries(Remapper${suite}Tests PRIVATE RemapperUI)
 add_test(NAME ${suite} COMMAND Remapper${suite}Tests)
endforeach()
add_executable(RemapperNotificationTests App/Tests/SnapshotNotificationTests.cpp)
target_link_libraries(RemapperNotificationTests PRIVATE user32)
add_test(NAME SnapshotNotification COMMAND RemapperNotificationTests)
add_executable(RemapperDriverSetupTests App/Tests/DriverSetupTests.cpp)
target_link_libraries(RemapperDriverSetupTests PRIVATE RemapperUI)
add_test(NAME DriverSetup COMMAND RemapperDriverSetupTests)
